# Dependency direction for hexuria/ultra-instinct (AGENTS.md "Crates (product graph)").
#
# Format:
#   <crate>: <allowed aui-* deps, space separated>
#       | <banned crates anywhere in the crate's dependency closure; `*` is a glob>
#
# Checked by scripts/check_architecture.py in CI:
#   1. every workspace member has an entry (new crates must declare their direction);
#   2. a crate's normal and build dependencies on workspace crates are a subset of "allowed";
#   3. no banned crate appears anywhere in its resolved dependency closure (incl. dev-deps).
#
# The product path is blocking and single-threaded by design (AGENTS.md anti-drift #10):
# no async runtime, HTTP client, or model SDK below aui-cli. typesafe-sdk is allowed
# only in aui-cli's closure (the optional `jev` example leg), so tokio / reqwest /
# hyper reach the workspace exclusively through that crate's gated edge.
aui-core:
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-geometry: aui-core
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-observe: aui-core
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-resonance: aui-core aui-geometry aui-hyper
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-protocol: aui-core
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-cdp:
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-browser: aui-cdp aui-core aui-geometry aui-observe aui-resonance
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-guard: aui-core aui-protocol aui-resonance
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-policy: aui-core
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-agent: aui-core aui-browser aui-guard aui-observe aui-policy aui-protocol
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-mcp: aui-protocol aui-core aui-resonance aui-browser aui-guard aui-observe
    | tokio axum sqlx reqwest hyper typesafe-sdk
aui-cli: aui-core aui-resonance aui-browser aui-guard aui-observe aui-mcp aui-protocol aui-agent aui-policy
    | axum sqlx
# Frozen experiment (HGRA). Leaf crate: no aui-* edges, no async/network deps.
aui-hyper:
    | tokio axum sqlx reqwest hyper typesafe-sdk
